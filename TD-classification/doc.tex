Nous observons des variables représentées par un vecteur $\vec{x} \in \R^d$
(par ex. l'âge et la pression sanguine), ainsi qu'un état binaire $y \in \{-1,1\}$
(par ex. être atteint d'une maladie ou non), sur $n$ individus :
$\{(\vec{x}_i,y_i)\}_{i = 1,\dots, n}$ est l'\emph{ensemble d'apprentissage}.
Or, on suppose l'existence d'une règle de décision linéaire, dépendant de
$\vec{w} \in \R^d$ et $b \in \R$, permettant de prédire $y$
en fonction de $\vec{x}$ :
\[
\forall i \in \{1,\dots,n\}, \ R_{\vec{w},b}(x_i) :=
\left\{
\begin{array}{ll}
  +1 & \quad \mathrm{si}\quad \vec{w}\cdot\vec{x}_i + b \geq 0 \\
  -1 & \quad \mathrm{sinon} \\
\end{array}
\right.
\]

\section{Séparateurs à vaste marge}
\label{sec:svm}

Le problème de classification décrit ci-dessus est celui considéré dans les
méthodes d'apprentissage connues sous le nom de \emph{Support Vector Machines}
ou \emph{Séparateurs à Vaste Marge} en français (SVM).
Ce sont des classifieurs binaires pour des données linéairement séparables\footnote{Ils
peuvent aussi être adaptés au cas où les données ne sont pas parfaitement séparables
ou au cas où les données sont non-linéairement séparables.} 
qui ont de bonnes capacités de généralisation parce qu'ils maximisent
la marge de séparation.

\begin{exercice}
  Considérons l'ensemble des contraintes d'inégalités suivantes :
  \[ \forall i \in \{1,\dots,n\}, \  y_i( \vec{w}\cdot\vec{x}_i + b ) > 0. \]
  Montrer que si $\vec{w}$ et $b$ respectent toutes ces contraintes alors
  le classifieur est correct à 100\%, c'est-à-dire que 
  $\forall i \in \{1,\dots,n\}, \ R_{\vec{w},b}(x_i) = y_i$. 
\end{exercice}

\begin{solution}
  Pour que le produit $y_i( \vec{w}\cdot\vec{x}_i + b )$
  soit strictement positif, il faut soit
  \begin{itemize}
  \item que $y_i = 1$ et $\vec{w}\cdot\vec{x}_i + b > 0$ ; 
    dans ce cas, $y_i = f(x_i) = 1$ (correct).
  \item que $y_i = -1$ et $\vec{w}\cdot\vec{x}_i + b < 0$ ;
    dans ce cas, $y_i = f(x_i) = -1$ (correct).
  \end{itemize}
\end{solution}

\begin{exercice}
  Considérons maintenant le problème suivant : 
\[
\text{(P)}
\left\{
\begin{array}{ll}
  \text{trouver} \ \vec{w} \in \R^{d}, b\in\R \ \text{qui minimisent} \ \|\vec{w}\|^2 \ \text{tels que} & \\
  \forall i \in \{1,\dots,n\}, \ y_i(\vec{w} \cdot \vec{x}_i + b) \geq 1. 
\end{array}
\right.
\]

Montrez qu'une solution $\vec{w}, b$ de (P)  fournit un classifieur
correct à $100\%$ sur l'ensemble d'apprentissage et maximise la marge de séparation.   
\end{exercice}

\begin{exercice}
  Montrer que si les données d'apprentissage sont linéairement séparables,
  alors il existe une solution au problème (P).
\end{exercice}

\section{Perceptron}
\label{sec:perceptron}

Le perceptron, ancêtre des réseaux de neurones, est un classifieur binaire pour des
données linéairement séparables. Il est associé à un algorithme qui permet de trouver,
à partir de l'ensemble d'apprentissage, les paramètres $\vec{w},b$ inconnus, appelés
respectivement \emph{poids} et \emph{biais}. Cet algorithme est une variante de la
descente de gradient pour minimiser une fonction objectif construite à partir des
contraintes non satisfaites.

\begin{exercice}
  Soient $E_{\vec{w},b} := \{i \in \{1,\dots,n\} \mid \ y_i (\vec{w}\cdot\vec{x}_i + b) \leq 0\}$,
  l'ensemble des contraintes non satisfaites et la fonction objectif suivante :
  \[
  Q(\vec{w},b) :=
  \quad
  \left\{
  \begin{array}{ll}
    - \sum_{i \in E_{\vec{w}},b} y_i (\vec{w}\cdot\vec{x}_i + b) & \text{si}\quad E_{\vec{w},b} \neq \emptyset \\
    0 & \text{sinon} \\
  \end{array}
  \right.
  \]

  \begin{enumerate}
  \item Expliquez pourquoi $Q$ est bornée inférieurement, c'est-à-dire que
    pour tout $\vec{w} \in \R^d, b \in \R$,  $Q(\vec{w},b) \geq 0$. 
  \item Calculez le gradient de $Q$\footnote{Pour vous aider :
  le gradient de $f(\vec{w}) = \vec{w}\cdot\vec{x}$ est $\nabla f(\vec{w}) = \vec{x}$}.
  \item En utilisant le schéma général de la descente de gradient,
    proposez une règle de mise à jour dans le cas où $E_{\vec{w},b} \neq \emptyset$. 
  \item Comme $Q$ est une somme de fonctions, une alternative à la descente de gradient classique
  consiste à mettre à jour la valeur de $\vec{w}$ et $b$ après chaque paire $(\vec{x}_i, y_i)$
  considérée. \'Ecrivez l'algorithme d'apprentissage complet mettant en oeuvre cette stratégie. 
  \end{enumerate}
\end{exercice}

\begin{solution}
  \begin{enumerate}
  \item Si $E_{\vec{w}} = \emptyset$, $Q(\vec{w}) = 0$.
    Sinon $Q(\vec{w}) = \sum_{i \in E_{\vec{w}}} - \big( y_i (\vec{w}\cdot\vec{x}_i) \big)$,
    ce qui est par définition de $E_{\vec{w}}$ une somme de termes $> 0$ et donc
    $Q(\vec{w}) > 0$ dans ce cas. 
  \item Si $E_{\vec{w}} = \emptyset$, ${\nabla Q}(\vec{w}) = \vec{0}$. Sinon,
    ${\nabla Q}(\vec{w}) = -y_i \vec{x}_i$.
  \item Pour une constante positive $\lambda > 0$, qu'on appelle dans ce
    contexte \emph{taux d'apprentissage}, la descente de gradient s'exprime
    comme :
    \[ \vec{w}^{(k+1)} = \vec{w}^{(k)} + \lambda \big( \sum_{i \in E_{\vec{w}}} y_i \vec{x}_i \big). \]
  \item Voir l'algorithme ci-après. 
  \end{enumerate}
    \begin{algorithm}[htbp]
    \caption{Algorithme d'apprentissage du perceptron}
    \KwIn{l'ensemble d'apprentissage $\{(\vec{x}_i,y_i)\}_{i = 1,\dots, n}$,\\
      le taux d'apprentissage $\lambda$}
    \KwOut{poids $\vec{w}$ du perceptron}
    $\vec{w} \leftarrow \vec{0}$ \;
    \Repeat{$\forall i, \ y_i(\vec{w}\cdot\vec{x}_i) \geq 0$}{
      \ForEach{$i = 1,\dots, n$}{
        \If{$y_i(\vec{w}\cdot\vec{x}_i) < 0$}{
          $\vec{w} \leftarrow \vec{w} + \lambda y_i\vec{x}_i$ \;
        }
      }
    }(\tcp*[h]{plus d'erreurs})
    \Return{$\vec{w}$} \;
  \end{algorithm}
\end{solution}
